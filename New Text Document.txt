import re
with open("app.py", "r", encoding="utf-8") as f:
    code = f.read()
# Xóa comment
code = re.sub(r'#.*', '', code)
# Xóa khoảng trắng thừa
code = "\n".join([line.strip() for line in code.split("\n") if line.strip()])
with open("app_clean.py", "w", encoding="utf-8") as f:
    f.write(code)
